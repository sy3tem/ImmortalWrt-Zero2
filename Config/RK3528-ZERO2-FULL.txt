#设备平台
CONFIG_TARGET_rockchip=y
CONFIG_TARGET_rockchip_armv8=y
#设备: NanoPi Zero2 (RK3528A) —— 上游原生支持
CONFIG_TARGET_DEVICE_rockchip_armv8_DEVICE_friendlyarm_nanopi-zero2=y

#=====================================================================
# Zero2 与 R28S 的硬件差异(决定了本文件与 RK3528-R28S-FULL.txt 的不同):
#   Zero2 : 单网口 gmac1(RGMII->RTL8211F), LED x2(SYS/LED1), 无板载WiFi
#           WiFi 走 M.2 Key-E (PCIe2.1), 需自插 RTL8822CE 模块
#   R28S  : 双网口(gmac1 + RTL8111H PCIe), LED x3, 板载 AIC8800(SDIO)
# 因此: 不引入 kmod-r8169(RTL8111H 专用) / kmod-aic8800-sdio(板载WiFi专用)
#=====================================================================

#WiFi: 不接无线模块, 不编任何 WiFi 驱动/固件/管理组件(用户明确要求)
#说明: Zero2 的 WiFi 走 M.2 Key-E(PCIe), 需自插模块; 本固件不使用无线,
#      故上游设备定义默认带的 kmod-rtw88-8822ce / wpad-openssl / wifi-scripts 全部不编.
#      注意: 若后续要用 LuCI 管理, 仍需 luci 本体(已在 GENERAL.txt 里), 但不会有"无线"菜单.
CONFIG_PACKAGE_kmod-rtw88-8822ce=n
CONFIG_PACKAGE_wpad-openssl=n
CONFIG_PACKAGE_wifi-scripts=n
#确保无线相关组件不被间接拉起
CONFIG_PACKAGE_wpad-basic-mbedtls=n
CONFIG_PACKAGE_wpad-basic-openssl=n
CONFIG_PACKAGE_wpad-full-mbedtls=n
CONFIG_PACKAGE_wpad-full-openssl=n
CONFIG_PACKAGE_kmod-cfg80211=n
CONFIG_PACKAGE_kmod-mac80211=n
CONFIG_PACKAGE_kmod-nl80211=n
CONFIG_PACKAGE_iw=n
CONFIG_PACKAGE_iwinfo=n

#分区: boot 128MB(内核+dtb) / rootfs 1024MB
CONFIG_TARGET_KERNEL_PARTSIZE=128
CONFIG_TARGET_ROOTFS_PARTSIZE=1024
CONFIG_TARGET_ROOTFS_EXT4FS=n

#科学插件: Open-Box (sing-box 一体化方案, 运行时 curl 安装, 非编译期软件包)
#说明: Open-Box 不属于 OpenWrt 软件包, 面板与 sing-box 内核由安装脚本自带,
#      因此固件侧【不编】passwall/xray 等代理软件包, 只预置其运行依赖.
#安装命令(root SSH):
#  curl -fsSL https://raw.githubusercontent.com/liandu2024/Open-Box/main/scripts/install.sh | sh
#国内网络不畅时改用镜像:
#  curl -fsSL https://gh-proxy.com/raw.githubusercontent.com/liandu2024/Open-Box/main/scripts/install.sh | sh -s -- --mirror
#默认面板端口 3036, 浏览器打开 http://<路由器IP>:3036
#--------------------------------------------------------------------
#Open-Box 运行依赖(按项目 README 列出的依赖清单):
#  kmod-tun / kmod-nft-queue / kmod-nft-nat / kmod-veth / ip-full / ca-bundle
CONFIG_PACKAGE_kmod-tun=y
CONFIG_PACKAGE_kmod-nft-queue=y
CONFIG_PACKAGE_kmod-nft-nat=y
CONFIG_PACKAGE_kmod-veth=y
CONFIG_PACKAGE_ip-full=y
CONFIG_PACKAGE_ca-bundle=y
#nftables 基础(Open-Box 用 nft 做入口旁路与 DNS 劫持)
CONFIG_PACKAGE_nftables-json=y
CONFIG_PACKAGE_kmod-nf-nathelper=y
CONFIG_PACKAGE_kmod-nf-nathelper-extra=y
